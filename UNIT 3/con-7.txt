7. Searching in a BST



#include <stdio.h>
#include <stdlib.h>
struct Node { int data; struct Node *left,*right; };
struct Node *insert(struct Node *r,int x) {
    if(!r) { r=malloc(sizeof(struct Node)); r->data=x; r->left=r->right=NULL; return r; }
    if(x<r->data) r->left=insert(r->left,x);
    else if(x>r->data) r->right=insert(r->right,x);
    return r;
}
struct Node *search(struct Node *r,int key) {
    if(!r || r->data==key) return r;
    if(key<r->data) return search(r->left,key);
    return search(r->right,key);
}
int main() {
    struct Node *root=NULL;
    root=insert(root,50); root=insert(root,30); root=insert(root,70);
    root=insert(root,20); root=insert(root,40);
    if(search(root,40)) printf("Element found");
    else printf("Element not found");
    return 0;
}



Output
Element found
