// 9. Reverse a String Using Stack


Every character is pushed into the stack. Characters are then popped one by one, producing the reverse string.


#include <stdio.h>
#include <string.h>

#define MAX 100

int main()
{
    char stack[MAX];
    int top = -1;

    char str[] = "REVERSE";

    for (int i = 0; i < strlen(str); i++)
    {
        stack[++top] = str[i];
    }

    printf("Original string: %s\n", str);

    printf("Reversed string: ");

    while (top >= 0)
    {
        printf("%c", stack[top]);
        top--;
    }

    printf("\n");

    return 0;
}
