// 11. Evaluation of Postfix Expression


In postfix evaluation, operands are pushed into the stack. When an operator is found, the required operands are popped, the operation is performed, and the result is pushed back.


#include <stdio.h>
#include <ctype.h>
#include <math.h>

#define MAX 100

int stack[MAX];
int top = -1;

void push(int value)
{
    stack[++top] = value;
}

int pop()
{
    return stack[top--];
}

int main()
{
    char postfix[] = "562-32^*4/+";

    for (int i = 0; postfix[i] != '\0'; i++)
    {
        char ch = postfix[i];

        if (isdigit(ch))
        {
            push(ch - '0');
        }
        else
        {
            int operand2 = pop();
            int operand1 = pop();
            int result;

            switch (ch)
            {
                case '+':
                    result = operand1 + operand2;
                    break;

                case '-':
                    result = operand1 - operand2;
                    break;

                case '*':
                    result = operand1 * operand2;
                    break;

                case '/':
                    result = operand1 / operand2;
                    break;

                case '^':
                    result = (int)pow(operand1, operand2);
                    break;

                default:
                    result = 0;
            }

            push(result);
        }
    }

    printf("Result = %d\n", pop());

    return 0;
}
