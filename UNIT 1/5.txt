// 5. Stack PUSH Operation


PUSH is used to insert an element into the stack. If the stack is full, stack overflow occurs.


#include <stdio.h>
#define MAX 5

int stack[MAX];
int top = -1;

void push(int item)
{
    if (top == MAX - 1)
    {
        printf("STACK OVERFLOW\n");
    }
    else
    {
        top++;
        stack[top] = item;
        printf("%d pushed into stack\n", item);
    }
}

int main()
{
    push(10);
    push(20);
    push(30);

    return 0;
}
