// 4. Merging Two Arrays

 
Merging combines the elements of two arrays into a third array. The elements of the first array are copied first, 
followed by the elements of the second array.

#include <stdio.h>

int main() {
    int a[] = {10, 20, 30};
    int b[] = {40, 50, 60};
    int c[6];

    int n = 3, m = 3;
    int k = 0;

    for(int i = 0; i < n; i++) {
        c[k] = a[i];
        k++;
    }

    for(int i = 0; i < m; i++) {
        c[k] = b[i];
        k++;
    }

    printf("Merged Array:\n");

    for(int i = 0; i < n + m; i++) {
        printf("%d ", c[i]);
    }

    return 0;
}
