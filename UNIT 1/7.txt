// 7. Stack DISPLAY Operation


DISPLAY prints stack elements from TOP to the bottom.


#include <stdio.h>

#define MAX 5

int stack[MAX];
int top = -1;

void push(int item)
{
    if (top == MAX - 1)
        printf("STACK OVERFLOW\n");
    else
        stack[++top] = item;
}

void display()
{
    if (top == -1)
    {
        printf("STACK EMPTY\n");
    }
    else
    {
        printf("Stack elements:\n");

        for (int i = top; i >= 0; i--)
        {
            printf("%d\n", stack[i]);
        }
    }
}

int main()
{
    push(10);
    push(20);
    push(30);

    display();

    return 0;
}
