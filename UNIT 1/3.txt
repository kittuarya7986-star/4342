// 3. Program to Multiply Two Matrices


In deletion, an element is removed from a specified position. The elements after that position are shifted one 
position to the left.

#include <stdio.h>
int main() {
    int a[] = {10, 20, 30, 40, 50};
    int n = 5;
    int pos = 2;

    for(int i = pos; i < n - 1; i++) {
        a[i] = a[i + 1];
    }

    n--;

    printf("Array after deletion:\n");

    for(int i = 0; i < n; i++) {
        printf("%d ", a[i]);
    }

    return 0;
}
