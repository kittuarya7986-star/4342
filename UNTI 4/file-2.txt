2. Binary Search

Binary Search works only on a sorted array. It repeatedly divides the search interval into two halves.


#include <stdio.h>
int main() {
    int a[] = {10, 20, 30, 40, 50, 60, 70};
    int n = 7, key = 50;
    int low = 0, high = n - 1, mid, found = 0;

    while(low <= high) {
        mid = (low + high) / 2;
        if(a[mid] == key) {
            printf("Element found at index %d", mid);
            found = 1;
            break;
        } else if(key < a[mid]) high = mid - 1;
        else low = mid + 1;
    }
    if(!found) printf("Element not found");
    return 0;
}


Output
Element found at index 4
