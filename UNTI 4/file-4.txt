4. Selection Sort
Selection Sort repeatedly finds the smallest element from the unsorted part and places it at the beginning of that unsorted part.


#include <stdio.h>
int main() {
    int a[] = {64, 25, 12, 22, 11}, n = 5;
    int i, j, min, temp;
    for(i = 0; i < n - 1; i++) {
        min = i;
        for(j = i + 1; j < n; j++)
            if(a[j] < a[min]) min = j;
        temp = a[i]; a[i] = a[min]; a[min] = temp;
    }
    printf("Sorted array: ");
    for(i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}


Output
Sorted array: 11 12 22 25 64
