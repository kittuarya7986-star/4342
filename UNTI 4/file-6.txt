6. Merge Sort
Merge Sort is a Divide and Conquer algorithm. It divides an array into smaller subarrays, sorts them, and then merges the sorted subarrays.



#include <stdio.h>

void merge(int a[], int l, int m, int r) {
    int i=l, j=m+1, k=0, temp[100];
    while(i<=m && j<=r) temp[k++] = (a[i]<a[j]) ? a[i++] : a[j++];
    while(i<=m) temp[k++] = a[i++];
    while(j<=r) temp[k++] = a[j++];
    for(i=0; i<k; i++) a[l+i] = temp[i];
}
void mergeSort(int a[], int l, int r) {
    if(l < r) {
        int m=(l+r)/2;
        mergeSort(a,l,m);
        mergeSort(a,m+1,r);
        merge(a,l,m,r);
    }
}
int main() {
    int a[] = {38, 27, 43, 3, 9, 82, 10}, n=7, i;
    mergeSort(a,0,n-1);
    printf("Sorted array: ");
    for(i=0;i<n;i++) printf("%d ",a[i]);
    return 0;
}


Output
Sorted array: 3 9 10 27 38 43 82
