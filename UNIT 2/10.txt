10. Deletion from the End



void del_last(struct node **head) {
    if (*head == NULL) return;

    if ((*head)->link == NULL) {
        free(*head);
        *head = NULL;
        return;
    }

    struct node *ptr = *head;
    while (ptr->link->link != NULL)
        ptr = ptr->link;

    free(ptr->link);
    ptr->link = NULL;
}
