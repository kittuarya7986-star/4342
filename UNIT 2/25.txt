25. Deletion from the Beginning of a Circular Singly Linked List



struct cnode *circularDeleteBeginning(struct cnode *head) {
    if (head == NULL) {
        return NULL;
    }
    if (head->next == head) {
        free(head);
        return NULL;
    }

    struct cnode *last = head;
    while (last->next != head) last = last->next;
    struct cnode *temp = head;
    head = head->next;
    last->next = head;
    free(temp);
    return head;
}
