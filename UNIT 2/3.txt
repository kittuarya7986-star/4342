3. Creating a Linked List Using User Input



#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *link;
};

int main() {
    struct node *head = NULL, *temp, *ptr;
    int n, i;
    printf("Enter number of elements: ");
    scanf("%d", &n);

    for (i = 0; i < n; i++) {
        temp = malloc(sizeof(struct node));
        if (temp == NULL) return 1;
        printf("Enter element %d: ", i + 1);
        scanf("%d", &temp->data);
        temp->link = NULL;

        if (head == NULL) {
            head = temp;
        }
        else {
            ptr = head;
            while (ptr->link != NULL) ptr = ptr->link;
            ptr->link = temp;
        }
    }

    printf("Linked List: ");
    ptr = head;
    while (ptr != NULL) {
        printf("%d -> ", ptr->data);
        ptr = ptr->link;
    }
    printf("NULL\n");

    while (head != NULL) {
        temp = head; head = head->link; free(temp);
    }
    return 0;
}
