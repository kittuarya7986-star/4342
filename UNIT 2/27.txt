27. Simple Queue Using an Array



#include <stdio.h>
#define MAX 5
int queue[MAX];
int front = -1, rear = -1;

void enqueue(int value) {
    if (rear == MAX - 1) {
        printf("Queue Overflow\n");
        return;
    }
    if (front == -1) front = 0;
    queue[++rear] = value;
}

void display(void) {
    if (front == -1) {
        printf("Queue is empty");
        return;
    }
    for (int i = front; i <= rear; i++) printf("%d ", queue[i]);
}

int main(void) {
    enqueue(10); enqueue(20); enqueue(30);
    display();
    return 0;
}
