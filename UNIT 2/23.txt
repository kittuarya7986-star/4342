23. Insertion at the Beginning of a Circular Singly Linked List




struct cnode *circularInsertBeginning(struct cnode *head, int data) {
    struct cnode *temp = malloc(sizeof(struct cnode));
    if (temp == NULL) {
        return head;
    }
    temp->data = data;

    if (head == NULL) {
        temp->next = temp;
        return temp;
    }

    struct cnode *last = head;
    while (last->next != head) last = last->next;
    temp->next = head;
    last->next = temp;
    return temp;
}
