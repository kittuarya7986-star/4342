7. Insertion at the End


void add_at_end(struct node **head, int data) {
    struct node *temp = malloc(sizeof(struct node));
    if (temp == NULL) return;
    temp->data = data;
    temp->link = NULL;

    if (*head == NULL) {
        *head = temp;
        return;
    }

    struct node *ptr = *head;
    while (ptr->link != NULL)
        ptr = ptr->link;
    ptr->link = temp;
}
