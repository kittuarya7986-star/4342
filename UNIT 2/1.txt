1. Creating a Node in C


A node contains data and a pointer to the next node.


#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *link;
};

int main() {
    struct node *head = malloc(sizeof(struct node));
    if (head == NULL) {
        return 1;
    }

    head->data = 45;
    head->link = NULL;

    printf("%d", head->data);
    free(head);
    return 0;
}
