2. Creating a Singly Linked List with Three Nodes


#include <stdio.h>
#include <stdlib.h>

struct node {
    int data;
    struct node *link;
};

int main() {
    struct node *head = malloc(sizeof(struct node));
    struct node *current = malloc(sizeof(struct node));
    struct node *current2 = malloc(sizeof(struct node));
    if (!head || !current || !current2) {
        return 1;
    }

    head->data = 45;
    head->link = current;
    current->data = 98;
    current->link = current2;
    current2->data = 3;
    current2->link = NULL;

    struct node *ptr = head;
    while (ptr != NULL) {
        printf("%d ", ptr->data);
        ptr = ptr->link;
    }

    free(current2);
    free(current);
    free(head);
    return 0;
}
